USE [sqlmi-prd-dp-cex-wfm-db];

WITH OCC_Complaints AS (
    SELECT CASE_NUMBER
    FROM reports.vw_OCC_Complaints

    UNION

    SELECT CASE_NUMBER
    FROM reports.vw_OCC_Complaint_Cases
),

CP_Unique AS (
    SELECT
        PREMISE,
        MAX(CONTRACT_ACCOUNT) AS CONTRACT_ACCOUNT,
        MAX(PSR_FLAG) AS PSR_FLAG,
        MAX(PRIMARY_CIRCUMSTANCE) AS PRIMARY_CIRCUMSTANCE,
        MAX(INDEX_DEPRIVATION) AS INDEX_DEPRIVATION,
        MAX(PROPERTY_TYPE) AS PROPERTY_TYPE,
        MAX(No_of_occupiers) AS No_of_occupiers,
        MAX(FLG_RESIDENTIAL) AS FLG_RESIDENTIAL,
        MAX(FLG_WATER_DEPENDANT) AS FLG_WATER_DEPENDANT,
        MAX(FLG_HELP_DURING_SUPPLY_INTERRUPTION)
            AS FLG_HELP_DURING_SUPPLY_INTERRUPTION
    FROM [playground-retail-db].[202610].[CP_DATA_V1]
    WHERE PREMISE IS NOT NULL
      AND PREMISE <> '1000000000'
    GROUP BY PREMISE
    HAVING COUNT(*) = 1
)

SELECT
    -- Case identification
    c.CASE_ID,
    c.CASE_NUMBER,
    c.ACCOUNT_ID,
    c.CONTACT_ID,

    -- Complaint indicators
    c.FLG_COMPLAINTCASE,

    CASE
        WHEN o.CASE_NUMBER IS NOT NULL THEN 1
        ELSE 0
    END AS IS_OCC_COMPLAINT,

    CASE
        WHEN c.FLG_COMPLAINTCASE = 1
          OR o.CASE_NUMBER IS NOT NULL
        THEN 1 ELSE 0
    END AS IS_COMPLAINT,

    -- Complaint classification
    c.FK_COMPLAINT_CATEGORY_HIGH_ID,
    c.FK_COMPLAINT_CATEGORY_LOW_ID,
    c.FK_COMPLAINT_STAGE_ID,
    c.FK_MID_LEVEL_WORK_TYPE_ID,
    c.FK_LOW_LEVEL_WORK_TYPE_ID,
    c.FK_REGION_ID,
    c.FK_CUSTOMER_SENTIMENT_ID,

    -- Lifecycle
    c.CREATED_DATE,
    c.CLOSED_DATE,
    c.FLG_CLOSED,
    c.FLG_ESCALATED,

    CASE
        WHEN c.CLOSED_DATE IS NOT NULL
        THEN DATEDIFF(
            HOUR, c.CREATED_DATE, c.CLOSED_DATE
        )
    END AS RESOLUTION_HOURS,

    -- Geography
    c.POSTCODE,
    a.EXTERNAL_ID AS ADDRESS_EXTERNAL_ID,
    cp.PREMISE,

    -- Customer demographics
    cp.CONTRACT_ACCOUNT,
    cp.PSR_FLAG,
    cp.PRIMARY_CIRCUMSTANCE,
    cp.INDEX_DEPRIVATION,
    cp.PROPERTY_TYPE,
    cp.No_of_occupiers,
    cp.FLG_RESIDENTIAL,
    cp.FLG_WATER_DEPENDANT,
    cp.FLG_HELP_DURING_SUPPLY_INTERRUPTION,

    -- Demographic match indicator
    CASE
        WHEN cp.PREMISE IS NOT NULL THEN 1
        ELSE 0
    END AS HAS_CP_DATA

FROM dbo.E_CASE_DETAILS c

LEFT JOIN OCC_Complaints o
    ON c.CASE_NUMBER = o.CASE_NUMBER

LEFT JOIN dbo.E_ADDRESS_LOCATION a
    ON c.ADDRESSLOCATION_ID = a.ADDRESS_LOCATION_ID

LEFT JOIN CP_Unique cp
    ON LTRIM(RTRIM(a.EXTERNAL_ID))
     = LTRIM(RTRIM(cp.PREMISE))

WHERE c.CREATED_DATE >= '2025-01-01';



SELECT
    FK_CUSTOMER_SENTIMENT_ID,
    FLG_COMPLAINTCASE,
    COUNT(*) AS TOTAL_CASES
FROM dbo.E_CASE_DETAILS
WHERE CREATED_DATE >= '2025-01-01'
GROUP BY
    FK_CUSTOMER_SENTIMENT_ID,
    FLG_COMPLAINTCASE
ORDER BY TOTAL_CASES DESC;

FROM dbo.E_CASE_DETAILS
WHERE CREATED_DATE >= '2025-01-01';


SELECT

        AS HAS_LOW_CATEGORY,

    COUNT(FK_MID_LEVEL_WORK_TYPE_ID)
        AS HAS_MID_WORK_TYPE,

    COUNT(FK_LOW_LEVEL_WORK_TYPE_ID)
        AS HAS_LOW_WORK_TYPE

FROM dbo.E_CASE_DETAILS
WHERE CREATED_DATE >= '2025-01-01';


SELECT
    FK_COMPLAINT_CATEGORY_HIGH_ID,
    FK_COMPLAINT_CATEGORY_LOW_ID,
    FK_MID_LEVEL_WORK_TYPE_ID,
    FK_LOW_LEVEL_WORK_TYPE_ID,
    COUNT(*) AS TOTAL_CASES
FROM dbo.E_CASE_DETAILS

GROUP BY
    FK_COMPLAINT_CATEGORY_HIGH_ID,
    FK_COMPLAINT_CATEGORY_LOW_ID,
    FK_MID_LEVEL_WORK_TYPE_ID,

SELECT TOP 20
    FK_MID_LEVEL_WORK_TYPE_ID,
    FK_LOW_LEVEL_WORK_TYPE_ID,
    COUNT(*) AS TOTAL_CASES
FROM dbo.E_CASE_DETAILS
WHERE CREATED_DATE >= '2025-01-01'
GROUP BY
    FK_MID_LEVEL_WORK_TYPE_ID,
    FK_LOW_LEVEL_WORK_TYPE_ID
ORDER BY TOTAL_CASES DESC;





SELECT TOP 20
    c.FK_MID_LEVEL_WORK_TYPE_ID,
    c.FK_LOW_LEVEL_WORK_TYPE_ID,
    ll.LOW_LEVEL_WORK_TYPE_DESC,
    COUNT(*) AS TOTAL_CASES
FROM dbo.E_CASE_DETAILS c
LEFT JOIN dbo.D_LOW_LEVEL_WORK_TYPE ll
    ON c.FK_LOW_LEVEL_WORK_TYPE_ID =
       ll.LOW_LEVEL_WORK_TYPE_ID
WHERE c.CREATED_DATE >= '2025-01-01'
GROUP BY
    c.FK_MID_LEVEL_WORK_TYPE_ID,
    c.FK_LOW_LEVEL_WORK_TYPE_ID,
    ll.LOW_LEVEL_WORK_TYPE_DESC
ORDER BY TOTAL_CASES DESC;



SELECT
    COUNT(*) AS TOTAL_ROWS,
    COUNT(DISTINCT CENSUS_LSOA_CODE) AS UNIQUE_LSOAS,
    COUNT(DISTINCT POSTCODE_NO_SPACE) AS UNIQUE_POSTCODES,
    COUNT(*) - COUNT(DISTINCT CENSUS_LSOA_CODE)
        AS ADDITIONAL_ROWS_BEYOND_UNIQUE_LSOAS,
    SUM(
        CASE
            WHEN CENSUS_LSOA_CODE IS NULL THEN 1
            ELSE 0
        END
    ) AS MISSING_LSOA_CODES
FROM dpcex_dev.default.lsoa_geog_postcode;


SELECT
    CENSUS_LSOA_CODE,
    COUNT(*) AS POSTCODE_COUNT,
    COUNT(DISTINCT geography_code) AS DISTINCT_GEOGRAPHY_CODES
FROM dpcex_dev.default.lsoa_geog_postcode
GROUP BY CENSUS_LSOA_CODE
ORDER BY POSTCODE_COUNT DESC
LIMIT 20;



SELECT
    CENSUS_LSOA_CODE,
    COUNT(*) AS POSTCODE_COUNT,
    COUNT(DISTINCT `Population_Density:_Persons_per_square_kilometre_`)
        AS DISTINCT_POPULATION_DENSITY
FROM dpcex_dev.default.lsoa_geog_postcode
GROUP BY CENSUS_LSOA_CODE
HAVING COUNT(DISTINCT `Population_Density:_Persons_per_square_kilometre_`) > 1
LIMIT 20;




CREATE OR REPLACE TEMP VIEW census_lsoa_master AS
SELECT *
FROM (
    SELECT
        *,
        ROW_NUMBER() OVER (
            PARTITION BY CENSUS_LSOA_CODE
            ORDER BY POSTCODE_NO_SPACE
        ) AS rn
    FROM dpcex_dev.default.lsoa_geog_postcode
) t
WHERE rn = 1;



SELECT
    COUNT(*) AS TOTAL_ROWS,
    COUNT(DISTINCT CENSUS_LSOA_CODE) AS UNIQUE_LSOAS
FROM census_lsoa_master;



SELECT
    COUNT(*) AS TOTAL_LSOAS,
    COUNT(`Population_Density:_Persons_per_square_kilometre_`)
        AS POPULATION_DENSITY_AVAILABLE,
    COUNT(*) -
    COUNT(`Population_Density:_Persons_per_square_kilometre_`)
        AS POPULATION_DENSITY_MISSING,
    MIN(`Population_Density:_Persons_per_square_kilometre_`)
        AS MIN_POPULATION_DENSITY,
    MAX(`Population_Density:_Persons_per_square_kilometre_`)
        AS MAX_POPULATION_DENSITY,
    AVG(`Population_Density:_Persons_per_square_kilometre_`)
        AS AVG_POPULATION_DENSITY
FROM census_lsoa_master;
